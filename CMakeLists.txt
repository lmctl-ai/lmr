cmake_minimum_required(VERSION 3.22)
project(rcli VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(lmcpp CONFIG REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/R/worker.R")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/R/worker.R" RCLI_WORKER_SOURCE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(src/worker.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/worker.hpp" @ONLY)
add_executable(rcli src/main.cpp)
target_link_libraries(rcli PRIVATE lmcpp::core)
target_include_directories(rcli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(rcli PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
include(CTest)
if(BUILD_TESTING)
 add_test(NAME rcli_ops COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh $<TARGET_FILE:rcli>)
 set_tests_properties(rcli_ops PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME no_libR_boundary COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-no-libR.sh)
 add_test(NAME no_libR_binary COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-no-libR.sh $<TARGET_FILE:rcli>)
endif()
